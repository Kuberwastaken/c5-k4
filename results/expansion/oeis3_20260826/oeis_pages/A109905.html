%N a(n) = greatest prime of the form k*(n-k) + 1, 0 if no such prime exists.

%C k can take values from 1 to floor[n/2].

%C a(n)=0 for k = 1, 6, 30 and 54. Are there any others? - _Robert Israel_, Feb 23 2018

%C There are none for n up to 10^9. - _Mauro Fiorentini_, Jul 24 2023

%H Ivan Neretin, <a href="/A109905/b109905.txt">Table of n, a(n) for n = 1..10000</a>

%e a(15) = 37 as 1*14 +1 = 16, 2*13 +1 = 27 are composite but 3*12 +1= 37 is a prime.

%e a(6) = 0 as 1*5 +1=6, 2*4 +1=9, 3*3 +1 = 10 are all composite.

%p f:= proc(n) local k;

%p for k from floor(n/2) to 1 by -1 do

%p if isprime(k*(n-k)+1) then return k*(n-k)+1 fi

%p od:

%p 0 end proc:

%p map(f, [$1..100]); # _Robert Israel_, Feb 23 2018

%t Table[Max@Prepend[Select[Table[k (n - k) + 1, {k, n/2}], PrimeQ], 0], {n, 60}] (* _Ivan Neretin_, Feb 23 2018 *)

%o (PARI) { a(n) = forstep(k=n\2,1,-1,if(isprime(k*(n-k)+1),return(k*(n-k)+1)));return(0) } \\ _Max Alekseyev_, Oct 04 2005
