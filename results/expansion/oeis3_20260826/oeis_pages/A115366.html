A115366
a(n) = the number of values of k <= 10^n such that sqrt(k*(k+1)*(k+2)*(k+3)+1) is prime.

%N a(n) = the number of values of k <= 10^n such that sqrt(k*(k+1)*(k+2)*(k+3)+1) is prime.

%C sqrt(k*(k+1)*(k+2)*(k+3)+1) = k^2 + 3*k + 1.

%C a(n)/A006880(n) ~= 1.78, 1.78, 1.7769, 1.7752, 1.7738, 1.7731. Conjecture: a(n)/A006880(n) -> 1.77...

%t c = 0; k = 1; Do[ While[k <= 10^n, If[ PrimeQ@ Round@ Sqrt[k(k + 1)(k + 2)(k + 3) + 1], c++ ]; k++ ]; Print@c, {n, 0, 9}] (* _Robert G. Wilson v_, May 01 2006 *)

%o (PARI) g(n)= { for(j=0,n, c=0; for(x=0,10^j, y=round(sqrt(x*(x+1)*(x+2)*(x+3)+1)); if(ispseudoprime(y),c++)); print1(c",") ) }

%o (Python)

%o from sympy import isprime

%o def A115366(n): return sum(1 for k in range(1,10**n+1) if isprime(k*(k+3)+1)) # _Chai Wah Wu_, Jun 19 2024
