%p H:= proc(n) H(n):= 1/n +`if`(n=1, 0, H(n-1)) end:

%p L:= proc(n) L(n):= ilcm(n, `if`(n=1, 1, L(n-1))) end:

%p a:= n-> L(n)/denom(H(n)):

%p seq(a(n), n=1..100); # _Alois P. Heinz_, Aug 30 2012
