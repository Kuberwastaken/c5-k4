%N Digital sum of the Fermat number 2^(2^n) + 1.

%C a(0), a(1), a(5), a(6), a(7) and a(11) are primes. Are there any more?

%F a(n) = ~4.5*A057755. - _Robert G. Wilson v_, Jul 02 2005

%e a(5)=59 because 2^(2^5) + 1 = 4294967297 and the sum of those decimal digits is 59.

%p a:=proc(n) local nn, z: nn:=convert(2^(2^n)+1,base,10): z:=nops(nn): add(nn[j],j=1..z) end: seq(a(n),n=0..22); # _Emeric Deutsch_

%t f[n_] := Plus @@ IntegerDigits[2^(2^n)] + 1; Table[ f[n], {n, 0, 27}] (* _Robert G. Wilson v_, Jul 02 2005 *)
